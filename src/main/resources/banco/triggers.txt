-- 1. Cria tabela para guardar o histórico (Auditoria)
create table if not exists audit_logs (
  id bigint generated by default as identity primary key,
  table_name text,
  record_id bigint,
  old_status text,
  new_status text,
  changed_at timestamp default now()
);

-- 2. Função que salva o histórico quando algo muda
create or replace function audit_ticket_status_change()
returns trigger
language plpgsql
as $$
begin
  -- Só grava se o status mudou (Ex: de ATIVO para USADO)
  if old.status is distinct from new.status then
    insert into audit_logs (table_name, record_id, old_status, new_status)
    values ('tickets', new.id, old.status, new.status);
  end if;
  return new;
end;
$$;

-- 3. O GATILHO DE AUDITORIA (Este é o Trigger 1)
create trigger trg_audit_tickets
after update on tickets
for each row
execute function audit_ticket_status_change();

-- 4. Função que impede fraude (Integridade)
create or replace function prevent_ticket_reactivation()
returns trigger
language plpgsql
as $$
begin
  -- Se tentar mudar de USADO/CANCELADO para ATIVO, o banco bloqueia
  if old.status IN ('USADO', 'CANCELADO') and new.status = 'ATIVO' then
    raise exception 'ERRO DE INTEGRIDADE: Não é permitido reativar ingresso usado/cancelado!';
  end if;
  return new;
end;
$$;

-- 5. O GATILHO DE INTEGRIDADE (Este é o Trigger 2)
create trigger trg_check_ticket_integrity
before update on tickets
for each row
execute function prevent_ticket_reactivation();
